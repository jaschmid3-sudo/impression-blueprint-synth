// Improved Bo Diddley Drum Voice with low-end resonance and slap function playDrum(time, isAccent, isSlap = false) { // Sub-kick / Tom resonance const osc = audioCtx.createOscillator(); const gain = audioCtx.createGain(); osc.frequency.setValueAtTime(isAccent ? 140 : 100, time); osc.frequency.exponentialRampToValueAtTime(40, time + 0.15); gain.gain.setValueAtTime(isAccent ? 0.9 : 0.4, time); gain.gain.exponentialRampToValueAtTime(0.001, time + 0.15); osc.connect(gain); gain.connect(audioCtx.destination); osc.start(time); osc.stop(time + 0.15); // Accent Slap / Rimshot Layer for Bo Diddley syncopation if (isSlap) { const noiseBuffer = audioCtx.createBuffer(1, audioCtx.sampleRate * 0.05, audioCtx.sampleRate); const output = noiseBuffer.getChannelData(0); for (let i = 0; i < noiseBuffer.length; i++) { output[i] = Math.random() * 2 - 1; } const whiteNoise = audioCtx.createBufferSource(); whiteNoise.buffer = noiseBuffer; const filter = audioCtx.createBiquadFilter(); filter.type = 'highpass'; filter.frequency.setValueAtTime(1000, time); const noiseGain = audioCtx.createGain(); noiseGain.gain.setValueAtTime(0.3, time); noiseGain.gain.exponentialRampToValueAtTime(0.01, time + 0.05); whiteNoise.connect(filter); filter.connect(noiseGain); noiseGain.connect(audioCtx.destination); whiteNoise.start(time); whiteNoise.stop(time + 0.05); } } // Sequence Generator with True Bo Diddley Clave function startSequence() { let tempo = 115; // Slightly peppier tempo for classic Bo Diddley feel let sixteenth = (60 / tempo) / 4; let currentStep = 0; // Authentic Bo Diddley 3:2 Clave Grid (16-step): // Bar 1 (3 hits): Step 0 (ONE), Step 3 (and of 2), Step 6 (FOUR) // Bar 2 (2 hits): Step 10 (TWO), Step 12 (THREE) const boDiddleyAccents = [0, 3, 6, 10, 12]; const fillerGhostHits = [2, 4, 8, 14]; // Secondary maraca/ghost tom taps // Boogie Woogie C-E-G pattern const bassPattern = [C3, C3, E3, E3, G3, G3, A3, Bb3]; // Blues Harmonica melody notes const harmonicaMelody = [ { note: C5, step: 0, dur: 2, bend: true }, { note: Bb4, step: 3, dur: 2, bend: false }, { note: G4, step: 6, dur: 2, bend: true }, { note: F4, step: 10, dur: 1.5, bend: true }, { note: Eb4, step: 12, dur: 2, bend: false } ]; function scheduleLoop() { let now = audioCtx.currentTime; for (let i = 0; i < 16; i++) { let time = now + (i * sixteenth); // Play True Bo Diddley Rhythm if (boDiddleyAccents.includes(i)) { // Main syncopated accents with slap layer playDrum(time, true, true); } else if (fillerGhostHits.includes(i)) { // Soft ghost hits to fill out the rumble playDrum(time, false, false); } // Play Boogie Bass (C-E-G walking) if (i % 2 === 0) { let bassIdx = (Math.floor(i / 2)) % bassPattern.length; playBassNote(bassPattern[bassIdx], time, sixteenth * 1.8); } // Play Harmonica Melody harmonicaMelody.forEach(m => { if (m.step === i) { playHarmonicaNote(m.note, time, m.dur * sixteenth * 4, m.bend); } }); } currentStep = (currentStep + 16) % 16; } scheduleLoop(); timerId = setInterval(scheduleLoop, sixteenth * 16 * 1000); }
